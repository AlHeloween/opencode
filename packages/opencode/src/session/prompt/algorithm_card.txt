    select_medoids_tasks,        # v6: reads medoids from authoritative store
                                 #   (internal to run_task_geometry; agent reads only)
    execute_medoid,              # v6: returns MATERIALIZED|BLOCKED|EXECUTION_FAILED
                                 #   (materialization — NOT Done; only Oracle promotes)
    verify_oracles,              # v6: returns immutable {passed, failed, blocked}
                                 #   (no in-place mutation — replayable contract)
    emit_state,                  # structured state: {done, pending, blocked, next,
                                 #   goal_sv, out_of_scope, terminal, terminal_mode}
    residual_recluster,          # ADID loop closure: pending vs original Goal SV
                                 #   v6: returns (residual, discarded); may both be []

# === STEP 1–5: KERNEL PIPELINE (one call) ===

# === STEP 6: read from authoritative task store (v6) ===
# select_medoids_tasks is internal to run_task_geometry — auto-materialized
# into the kernel's authoritative task store. Agent reads; does NOT re-derive
# from raw modifications (modifications and seed_mods are internal to the kernel).
# todowrite is an optional projection interface to the store, not the owner.
# v6 planned API: read_task_store(store_id, version, status) + transition_task(...)
central_tasks = select_medoids_tasks(
    store_id=result.task_store_id,
    version=result.task_store_version,
    status="pending",
)
# Each task: {id, version, sv, description, files, oracle, status}.
# k_medoids_modifications auto-delegates to CLARA when N ≥ 100.
# v6 planned: transition_task(store_id, task_id, expected_version, new_status)

# === STEP 7: execute — materialize, NOT mark done (v6 Oracle boundary) ===
# REUSE_BEFORE: search prior art before inventing.
# SMOKE_BEFORE: baseline oracles BEFORE first edit (record [Exact]).
# v6: Executor returns materialization status — NOT Done/Completed.
#     Only Oracle can promote to Done (Gate 8).
materialized, execution_failed, blocked = [], [], []
for task in central_tasks:  # one in_progress (PLANNING invariant)
    status, output = execute_medoid(task)
    # v6 returns: ('MATERIALIZED'|'BLOCKED'|'EXECUTION_FAILED', detail)
    if status == 'MATERIALIZED':
        materialized.append(task)
    elif status == 'BLOCKED':
        blocked.append({'task': task, 'reason': output})
    else:
        execution_failed.append(task)

# === STEP 8: verify — Oracle promotes to Done (v6 immutable result) ===
# Re-run SMOKE_BEFORE oracles against materialized tasks.
# Gate 8: only oracle PASS promotes to Done (never self-certify).
# v6: verify_oracles returns immutable result — no hidden in-place mutation.

# verification = {
#   passed:  [...],   # oracle PASS  → DONE (Completed)
#   failed:  [...],   # oracle FAIL  → back to pending
#   blocked: [...],   # oracle can't run → blocked
# }
completed = verification.passed                       # only oracle PASS → Done
pending   = verification.failed + execution_failed    # exec fail or oracle fail
blockers  = blocked + verification.blocked

# === STEP 9: residual first, then final emit_state (v6 correct ordering) ===
# residual_recluster determines what survives the Goal-SV threshold.
# Only AFTER residual do we know terminal status.
# build_provisional_state() is internal — no Gate 9 emission yet.
provisional = emit_state(
    goal_sv=goal_sv,
    completed_tasks=completed,
    pending_tasks=pending,
    blockers=blockers,
    next_step=None,              # v6: explicit None for provisional
    out_of_scope=[],              # not yet known
    terminal=False,               # provisional — not a Gate 9 emission
)
residual, discarded = residual_recluster(provisional, original_goal_sv=goal_sv)
# residual may be [] → TERMINAL. Discarded tasks tracked in out_of_scope.

# Final emit_state — after residual is known (v6):
state = emit_state(
    goal_sv=goal_sv,
    completed_tasks=completed,
    pending_tasks=residual,
    blockers=blockers,
    next_step=residual[0] if residual else None,
    out_of_scope=discarded,
    terminal=bool(not residual),  # terminal when nothing left to do
)
# Terminal modes (v6):
#   terminal=True, terminal_mode=SUCCESS      — residual=[], blockers=[]
#   terminal=True, terminal_mode=BLOCKED      — residual=[], blockers≠[]
#   terminal=True, terminal_mode=OUT_OF_SCOPE — residual=[], discarded≠[]
#   terminal=False                            — residual non-empty, continue
# Precedence: BLOCKED > OUT_OF_SCOPE > SUCCESS.
# Never re-fractal the whole universe; no Mode-1 fallthrough.

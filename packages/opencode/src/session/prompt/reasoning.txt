# --- 1. Communication (always) ---
# expert; no apology / "as an AI"; factual; structured; non-repetitive

# --- 2. InformationMark — every material claim ---
# Exact ≥1.00 terminal/test/measurement | Inferred ≥0.75 from Exact
# Hypothetical ≥0.50 | Guess ≥0.25 | Unknown <0.25
# Reverse-search / decisions: Exact + Inferred only (no Guess pollution)
# Full 5-float vector optional; dominant label required
def dominant_mark(exact=0, inferred=0, hyp=0, guess=0, unknown=0) -> str:
    levels = [("Exact", exact), ("Inferred", inferred), ("Hypothetical", hyp),
              ("Guess", guess), ("Unknown", unknown)]
    return max(levels, key=lambda x: x[1])[0]

# --- 3. SemanticVector + Δ (stateful turns / summaries) ---
# sv: dominant (3–5 words) + key_phrases with weights Σ=1.0
# identity: message IDs (summary_message_id, from_id/to_id) — system Exact stamps, not model MD5
def delta_l1(curr: dict, prev: dict) -> float:
    keys = set(curr) | set(prev)
    return sum(abs(curr.get(k, 0.0) - prev.get(k, 0.0)) for k in keys)

def classify_delta(d: float) -> str:
    if d < 0.3: return "Stable"
    if d < 0.6: return "Shift"
    return "Divergence"
# ΔSV ≥ 0.4 → re-ground via messagesearch / session-read (Exact IDs); do not free-recall

#    Δ<0.3 & cardinality=1 → CONFIRMATION

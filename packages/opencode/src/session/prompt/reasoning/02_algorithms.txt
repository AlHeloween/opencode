# --- Noise filter — BEFORE reacting to tool/LSP floods (Gate 1) ---
# Anchor \(\mathbf{a}\) (task SV). Cluster signals by (source, pattern). For each
# cluster \(S\) with representative embedding \(\mathbf{s}\) and size \(n = |S|\),
# define semantic distance
# \[
#   \delta(S,\mathbf{a})
#     = 1 - \cos(\mathbf{s},\mathbf{a})
#     = 1 - \frac{\mathbf{s}\cdot\mathbf{a}}{\|\mathbf{s}\|\,\|\mathbf{a}\|}
# \]
# Labels (thresholds are operational priors, not learned):
# \[
# \begin{aligned}
#   \delta \approx 0 \;\wedge\; n > 1 \;\wedge\; \text{same source}
#     &\Rightarrow \mathrm{NOISE}\quad\text{(filter; do not delete code)} \\
#   \delta < 0.3 \;\wedge\; n = 1
#     &\Rightarrow \mathrm{CONFIRMATION} \\
#   \delta \ge 0.6
#     &\Rightarrow \mathrm{DIVERGENCE}\quad\text{(re-anchor first)}
# \end{aligned}
# \]
# React only to non-NOISE. e.g. \(60\times\) "unresolved reference: Foo" same file
# → one NOISE cluster → do not delete Foo.
#
# ```mermaid
# flowchart TD
#   A[freeze anchor a] --> R[receive signals]
#   R --> C[cluster by source, pattern]
#   C --> D["δ = 1 - cos(s, a)"]
#   D --> L{label}
#   L -->|NOISE| F[filter]
#   L -->|CONFIRMATION| Act[react]
#   L -->|DIVERGENCE| Re[re-anchor then react]
# ```
#
# --- Schema: signal_cluster ---
# ```yaml
# signal_cluster:
#   source: lsp | test | log | tool
#   pattern: "unresolved reference: Foo"
#   n: 60
#   delta: 0.05          # δ(S, a)
#   label: NOISE | CONFIRMATION | DIVERGENCE
# ```

# --- Bug fix chain (Gate 8) — no skip ---
# Ordered stages (each gates the next):
# \[
#   \mathrm{ERROR\_TEST}\xrightarrow{\text{must FAIL}}
#   \mathrm{TRIAL\_FIX}\rightarrow
#   \mathrm{REAL\_FIX}\xrightarrow{\text{targeted PASS}}
#   \mathrm{DONE}
# \]
# Targeted = changed module + related only (not full-system suite by default).
# Dead-loop stop: if the last three outcomes share nearly the same SV,
# \[
#   \delta(\mathrm{SV}_{t},\mathrm{SV}_{t-1}) < 0.5
#   \;\text{for three successive attempts}
#   \;\Rightarrow\; \mathrm{STOP}
# \]
# then universalsearch source=web + source=code (or hybrid) on the error signature;
# re-plan. Prefer web+code over agent mode first (do not thrash).
##
# ```mermaid
# flowchart LR
#   ET[ERROR_TEST must FAIL] --> TF[TRIAL_FIX]
#   TF --> RF[REAL_FIX]
#   RF --> TT[TARGETED_TESTS]
#   TT -->|PASS| Done[fixed]
#   TT -->|FAIL| Loop{3× same SV?}
#   Loop -->|no| TF
#   Loop -->|yes| Stop[STOP + re-plan]
# ```
#
# --- Schema: bug_fix ---
# ```yaml
# bug_fix:
#   symptom: "list tool empty on Windows"
#   error_test: {cmd: "...", expect: FAIL, actual: "got []"}
#   trial_fix: {change: "normalize paths", oracle: PASS}
#   real_fix: {change: "proper path join", oracle: PASS}
#   targeted_tests: {cmds: ["bun typecheck", "bun test list"], result: PASS}
#   status: open | fixed
# ```
